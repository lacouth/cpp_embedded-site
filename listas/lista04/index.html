    Lista 04 · Máquinas de estados¶

    Exercícios do capítulo 4 — Máquinas de estados.

    Antes de escrever cada switch, desenhe a máquina num papel: estados em caixas, transições em setas, e a condição de cada seta escrita em cima dela. A maioria dos bugs desta lista aparece no desenho antes de aparecer no código.

    Como resolver

    Exercícios ex (sem placa): copie o bloco de código, cole no seu compilador online, preencha a parte marcada como ESCREVA SUA SOLUCAO AQUI e rode. A saída mostra quais testes passaram.

    Exercícios wk (no Wokwi): monte o projeto inicial no Wokwi, complete o código e confira o resultado contra o critério de pronto do enunciado.

    ex03¶

    Escreva atualiza() do Semaforo só com o tempo, sem pedestre (ignore pedido_ por enquanto): verde por TEMPO_VERDE_MS, amarelo por TEMPO_AMARELO_MS, vermelho por TEMPO_VERMELHO_MS, e de volta ao verde.

    Use um switch (fase_) com um case por fase, e troque de fase sempre pelo método privado entra(), que já acende o LED certo e marca o tempo de entrada em entrou_ms_.

    Semaforo s(25, 27, 26);
s.inicia();                     // verde em t=0
falso::avanca_tempo(5000);
s.atualiza();
s.fase();                       // -> Fase::Amarelo
falso::nivel(27);               // -> HIGH (amarelo aceso)

    // ex03 — versao de arquivo unico (gerada automaticamente,
// nao edite este arquivo direto: edite os fontes em exercicios/ e rode

#include <cassert>
#include <iostream>

// ============================================================
// NAO ALTERE — Arduino de mentira (no Wokwi, use #include <Arduino.h>)
// ============================================================
// arduino_falso.h -- um <Arduino.h> de mentira, para rodar no computador (ou num
// compilador online) o mesmo codigo de classe que roda no ESP32 do Wokwi.
//
// Nao e um simulador. E so o suficiente para:
//   * o compilador aceitar pinMode/digitalWrite/digitalRead/millis/Serial;
//   * os testes perguntarem "o que a classe fez com o pino?" e "o que ela
//     faria se o tempo andasse 500 ms?".
//
// Como funciona: cada pino guarda o ultimo nivel escrito e o ultimo modo
// configurado; millis() devolve um relogio que so anda quando o teste manda
// (falso::avanca_tempo). digitalRead devolve o que o teste pos no pino com
// falso::poe_nivel -- e assim que se "aperta um botao" sem botao.
//
// No Wokwi, troque  #include "arduino_falso.h"  por  #include <Arduino.h>
// e o resto do codigo continua igual.
#ifndef ARDUINO_FALSO_H
#define ARDUINO_FALSO_H

// ---- Constantes com os mesmos nomes do Arduino ----------------------------
const int LOW = 0;
const int HIGH = 1;

const int INPUT = 0;
const int OUTPUT = 1;
const int INPUT_PULLUP = 2;

// ---- Estado do "hardware" falso -------------------------------------------
namespace falso {

const int NUM_PINOS = 40;   // o ESP32 tem GPIO 0 a 39

struct Pino {
    int nivel = LOW;
    int modo = -1;          // -1: ninguem chamou pinMode ainda
    int escritas = 0;       // quantas vezes digitalWrite foi chamado neste pino
};

inline Pino& pino(int numero) {
    static Pino pinos[NUM_PINOS];
    return pinos[numero % NUM_PINOS];
}

inline unsigned long& relogio() {
    static unsigned long agora_ms = 0;
    return agora_ms;
}

// Para os testes: espiar e mexer no mundo externo.
inline int nivel(int numero) { return pino(numero).nivel; }
inline int modo(int numero) { return pino(numero).modo; }
inline int escritas(int numero) { return pino(numero).escritas; }
inline void poe_nivel(int numero, int nivel) { pino(numero).nivel = nivel; }
inline void avanca_tempo(unsigned long ms) { relogio() += ms; }

// Volta tudo ao estado de "placa recem-ligada".
inline void reinicia() {
    for (int i = 0; i < NUM_PINOS; i++) {
        pino(i) = Pino{};
    }
    relogio() = 0;
}

}  // namespace falso

// ---- As funcoes do Arduino --------------------------------------------------
inline void pinMode(int numero, int modo) {
    falso::pino(numero).modo = modo;
    // Com pull-up interno, um pino de entrada solto le HIGH -- igual ao real.
    if (modo == INPUT_PULLUP) {
        falso::pino(numero).nivel = HIGH;
    }
}

inline void digitalWrite(int numero, int nivel) {
    falso::pino(numero).nivel = nivel;
    falso::pino(numero).escritas++;
}

inline int digitalRead(int numero) {
    return falso::pino(numero).nivel;
}

inline unsigned long millis() {
    return falso::relogio();
}

inline void delay(unsigned long ms) {
    falso::relogio() += ms;
}

// Declaracoes do exercicio

#ifndef EX03_H
#define EX03_H

class Led {
private:
    const int pino_;
    bool ligado_;

public:
    explicit Led(int pino) : pino_(pino), ligado_(false) {}
    void inicia() { pinMode(pino_, OUTPUT); desliga(); }
    void liga() { digitalWrite(pino_, HIGH); ligado_ = true; }
    void desliga() { digitalWrite(pino_, LOW); ligado_ = false; }
    void alterna() { if (ligado_) { desliga(); } else { liga(); } }
    bool esta_ligado() const { return ligado_; }
};

const unsigned long TEMPO_VERDE_MS = 5000;
const unsigned long TEMPO_VERDE_MINIMO_MS = 1000;
const unsigned long TEMPO_AMARELO_MS = 1000;
const unsigned long TEMPO_VERMELHO_MS = 3000;

enum class Fase { Verde, Amarelo, Vermelho };

class Semaforo {
private:
    Led verde_;
    Led amarelo_;
    Led vermelho_;
    Fase fase_;
    unsigned long entrou_ms_;
    bool pedido_;

    void entra(Fase nova) {
        fase_ = nova;
        entrou_ms_ = millis();
        verde_.desliga();
        amarelo_.desliga();
        vermelho_.desliga();
        switch (nova) {
            case Fase::Verde:    verde_.liga();    break;
            case Fase::Amarelo:  amarelo_.liga();  break;
            case Fase::Vermelho: vermelho_.liga(); pedido_ = false; break;
        }
    }

public:
    Semaforo(int pino_verde, int pino_amarelo, int pino_vermelho)
        : verde_(pino_verde), amarelo_(pino_amarelo), vermelho_(pino_vermelho),
          fase_(Fase::Verde), entrou_ms_(0), pedido_(false) {}

    void inicia() {
        verde_.inicia();
        amarelo_.inicia();
        vermelho_.inicia();
        entra(Fase::Verde);
    }

    void pede_travessia() { pedido_ = true; }

    void atualiza();

    Fase fase() const { return fase_; }
};

#endif

// ESCREVA SUA SOLUCAO AQUI
// ============================================================
void Semaforo::atualiza() {

// ============================================================
// NAO ALTERE — testes
// ============================================================
// Avanca de 100 em 100 ms chamando atualiza, como um loop() lento.
void roda(Semaforo& s, unsigned long ms) {
    for (unsigned long t = 0; t < ms; t += 100) {
        falso::avanca_tempo(100);
        s.atualiza();
    }
}

int main() {
    falso::reinicia();
    Semaforo s(25, 27, 26);
    s.inicia();
    assert(s.fase() == Fase::Verde);
    assert(falso::nivel(25) == HIGH && falso::nivel(27) == LOW && falso::nivel(26) == LOW);

    roda(s, 4900);
    assert(s.fase() == Fase::Verde);
    roda(s, 100);                                  // 5000
    assert(s.fase() == Fase::Amarelo);
    assert(falso::nivel(25) == LOW && falso::nivel(27) == HIGH);

    roda(s, 900);
    assert(s.fase() == Fase::Amarelo);
    roda(s, 100);                                  // 6000
    assert(s.fase() == Fase::Vermelho);
    assert(falso::nivel(26) == HIGH && falso::nivel(27) == LOW);

    roda(s, 3000);                                 // 9000
    assert(s.fase() == Fase::Verde);
    assert(falso::nivel(25) == HIGH && falso::nivel(26) == LOW);

    // um ciclo completo dura 9 s
    roda(s, 9000);
    assert(s.fase() == Fase::Verde);

    // o tempo de cada fase conta a partir da ENTRADA nela
    roda(s, 5000);
    assert(s.fase() == Fase::Amarelo);
    roda(s, 500);
    assert(s.fase() == Fase::Amarelo);

    std::cout << "Todos os testes passaram!\n";
    return 0;
}

    ex04¶

    Agora com pedestre. Escreva atualiza() completo: igual ao exercício anterior, mas o verde também termina se houver pedido (pedido_) e o verde já durou pelo menos TEMPO_VERDE_MINIMO_MS. O pedido é zerado sozinho pelo entra() quando o semáforo entra no vermelho.

    s.inicia();                     // verde em t=0
falso::avanca_tempo(300);
s.pede_travessia();             // pedido cedo demais...
s.atualiza();
s.fase();                       // -> Fase::Verde (ainda nao deu 1 s)
falso::avanca_tempo(700);
s.atualiza();
s.fase();                       // -> Fase::Amarelo (1 s de verde, com pedido)

    Um pedido feito durante o amarelo é esquecido ao entrar no vermelho (a travessia já vai acontecer). Um pedido feito durante o vermelho fica guardado e encurta o próximo verde.

    ex05¶

    Escreva atualiza() do BotaoFiltrado, o debounce do capítulo, a partir do diagrama:

    Estado Se a leitura é... Vai para Observação
    Solto apertado TalvezApertado marca desde_ms_
    TalvezApertado solto Solto era quique
    TalvezApertado apertado há ESTAVEL_MS ou mais Apertado clique nesta volta
    Apertado solto TalvezSolto marca desde_ms_
    TalvezSolto apertado Apertado era quique
    TalvezSolto solto há ESTAVEL_MS ou mais Solto

    clique_ deve ser true só na volta da transição para Apertado: comece cada atualiza() zerando-o.

#ifndef BOTAO_FILTRADO_H
#define BOTAO_FILTRADO_H

#include <Arduino.h>

// Quanto tempo a leitura precisa ficar parada para ser levada a serio.
const unsigned long ESTAVEL_MS = 30;

enum class EstadoBotao { Solto, TalvezApertado, Apertado, TalvezSolto };

class BotaoFiltrado {
private:
    const int pino_;
    EstadoBotao estado_;
    unsigned long desde_ms_;   // quando a leitura mudou pela ultima vez
    bool clique_;              // houve clique NESTA volta?

public:
    explicit BotaoFiltrado(int pino)
        : pino_(pino), estado_(EstadoBotao::Solto), desde_ms_(0), clique_(false) {}

    void inicia() {
        pinMode(pino_, INPUT_PULLUP);
    }

    // Chame uma vez por volta do loop(), antes de perguntar qualquer coisa.
    void atualiza() {
        bool lido = (digitalRead(pino_) == LOW);
        unsigned long agora = millis();
        clique_ = false;

        switch (estado_) {
            case EstadoBotao::Solto:
                if (lido) {
                    estado_ = EstadoBotao::TalvezApertado;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezApertado:
                if (!lido) {
                    estado_ = EstadoBotao::Solto;          // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Apertado;       // firme: e clique
                    clique_ = true;
                }
                break;

            case EstadoBotao::Apertado:
                if (!lido) {
                    estado_ = EstadoBotao::TalvezSolto;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezSolto:
                if (lido) {
                    estado_ = EstadoBotao::Apertado;       // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Solto;
                }
                break;
        }
    }

    bool foi_apertado() const {
        return clique_;
    }

    bool esta_apertado() const {
        return estado_ == EstadoBotao::Apertado || estado_ == EstadoBotao::TalvezSolto;
    }

    wk02 · no Wokwi¶

    A lanterna no Wokwi. É o exercício 6 desta lista, agora com LED e botão de verdade (simulados, e quicando). Complete clique() e atualiza() em lanterna.h; o sketch já está pronto.

    Critério de pronto: o Monitor Serial começa com desligada. Cada aperto no botão A imprime exatamente uma linha com o próximo modo (ligada, piscando, desligada, …) e o LED vermelho obedece: apagado, aceso, piscando 4 vezes por segundo.

    Pergunta: a Lanterna não sabe que existe um botão, só recebe clique(). Que vantagem isso traz se amanhã o clique vier de um comando pelo Monitor Serial em vez de um botão?

    Projeto inicial — crie um projeto ESP32 no Wokwi e cole cada aba no arquivo de mesmo nome (veja como montar o projeto).

    // Lista 04 - wk02: a lanterna de um botao so. Complete lanterna.h.
// O botao A QUICA de proposito: o BotaoFiltrado cuida disso.
#include "botao_filtrado.h"
#include "lanterna.h"

BotaoFiltrado botao_a(32);
Lanterna lanterna(26);

void setup() {
    Serial.begin(115200);
    botao_a.inicia();
    lanterna.inicia();
    Serial.println(nome_do_modo(lanterna.modo()));
}

void loop() {
    botao_a.atualiza();
    if (botao_a.foi_apertado()) {
        lanterna.clique();
        Serial.println(nome_do_modo(lanterna.modo()));
    }
    lanterna.atualiza();
}

    // botao_filtrado.h -- debounce como maquina de estados (capitulo 4).
#ifndef BOTAO_FILTRADO_H
#define BOTAO_FILTRADO_H

#include <Arduino.h>

// Quanto tempo a leitura precisa ficar parada para ser levada a serio.
const unsigned long ESTAVEL_MS = 30;

enum class EstadoBotao { Solto, TalvezApertado, Apertado, TalvezSolto };

class BotaoFiltrado {
private:
    const int pino_;
    EstadoBotao estado_;
    unsigned long desde_ms_;   // quando a leitura mudou pela ultima vez
    bool clique_;              // houve clique NESTA volta?

public:
    explicit BotaoFiltrado(int pino)
        : pino_(pino), estado_(EstadoBotao::Solto), desde_ms_(0), clique_(false) {}

    void inicia() {
        pinMode(pino_, INPUT_PULLUP);
    }

    // Chame uma vez por volta do loop(), antes de perguntar qualquer coisa.
    void atualiza() {
        bool lido = (digitalRead(pino_) == LOW);
        unsigned long agora = millis();
        clique_ = false;

        switch (estado_) {
            case EstadoBotao::Solto:
                if (lido) {
                    estado_ = EstadoBotao::TalvezApertado;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezApertado:
                if (!lido) {
                    estado_ = EstadoBotao::Solto;          // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Apertado;       // firme: e clique
                    clique_ = true;
                }
                break;

            case EstadoBotao::Apertado:
                if (!lido) {
                    estado_ = EstadoBotao::TalvezSolto;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezSolto:
                if (lido) {
                    estado_ = EstadoBotao::Apertado;       // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Solto;
                }
                break;
        }
    }

    bool foi_apertado() const {
        return clique_;
    }

    bool esta_apertado() const {
        return estado_ == EstadoBotao::Apertado || estado_ == EstadoBotao::TalvezSolto;
    }
};

#endif

#ifndef LED_H
#define LED_H

#include <Arduino.h>

class Led {
private:
    const int pino_;
    bool ligado_;

public:
    explicit Led(int pino) : pino_(pino), ligado_(false) {}

    void inicia() {
        pinMode(pino_, OUTPUT);
        desliga();
    }

    void liga() {
        digitalWrite(pino_, HIGH);
        ligado_ = true;
    }

    void desliga() {
        digitalWrite(pino_, LOW);
        ligado_ = false;
    }

    void alterna() {
        if (ligado_) {
            desliga();
        } else {
            liga();
        }
    }

    bool esta_ligado() const {
        return ligado_;
    }
};

#endif

#ifndef TEMPORIZADOR_H
#define TEMPORIZADOR_H

#include <Arduino.h>

// Avisa, sem travar o programa, que um intervalo de tempo passou.
// Periodico: depois de avisar, comeca a contar o proximo intervalo sozinho.
class Temporizador {
private:
    unsigned long intervalo_ms_;
    unsigned long inicio_ms_;

public:
    explicit Temporizador(unsigned long intervalo_ms)
        : intervalo_ms_(intervalo_ms), inicio_ms_(0) {}

    // Comeca a contar a partir de agora.
    void reinicia() {
        inicio_ms_ = millis();
    }

    // true uma vez a cada intervalo. Chame a cada volta do loop().
    bool expirou() {
        unsigned long agora = millis();
        if (agora - inicio_ms_ >= intervalo_ms_) {
            inicio_ms_ = agora;
            return true;
        }
        return false;
    }
};

#endif

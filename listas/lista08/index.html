    Lista 08 · Interfaces e padrões de testabilidade¶

    Exercícios do capítulo 8 — Interfaces, HAL e padrões de testabilidade.

    Cada exercício traz, no .h, as interfaces de que precisa (ISaida, ISensor, IRelogio), iguais às de projetos/esqueleto/logica/interfaces.h. Os exercícios 1 a 4 constroem as peças (uma saída de verdade e três dublês); os 5 a 8 usam essas peças para testar a lógica da bomba d'água sem placa e sem esperar. Nenhuma classe usa new: os dublês guardam o que precisam em arrays de tamanho fixo.

    Como resolver

    Exercícios ex (sem placa): copie o bloco de código, cole no seu compilador online, preencha a parte marcada como ESCREVA SUA SOLUCAO AQUI e rode. A saída mostra quais testes passaram.

    Exercícios wk (no Wokwi): monte o projeto inicial no Wokwi, complete o código e confira o resultado contra o critério de pronto do enunciado.

    ex01¶

    Complete a SaidaDigital, a implementação de ISaida que liga e desliga um pino. Ela serve a um LED (ativo em nível alto) e a um módulo relé ativo em nível baixo, que liga com LOW:

    • inicia() configura o pino como saída e deixa a saída desligada;
    • liga() e desliga() escrevem o nível certo para a polaridade e lembram o estado;

    Os testes usam a saída pela interface (ISaida&), como a lógica vai usar.

rele.inicia();

ISaida& s = rele;
s.liga();
falso::nivel(4);                // -> LOW
s.ligada();                     // -> true

// manda (falso::avanca_tempo em ms, falso::avanca_micros em us). digitalRead devolve o que o teste pos no pino com

#ifndef ARDUINO_FALSO_H
#define ARDUINO_FALSO_H

// ---- Constantes com os mesmos nomes do Arduino ----------------------------
const int LOW = 0;
const int HIGH = 1;

const int INPUT = 0;
const int OUTPUT = 1;
const int INPUT_PULLUP = 2;

// Atenuacao do ADC, com os nomes do core 3.x (o padrao do chip e ADC_11db).
enum adc_attenuation_t { ADC_0db, ADC_2_5db, ADC_6db, ADC_11db, ADC_ATTENDB_MAX };

// ---- Estado do "hardware" falso -------------------------------------------
namespace falso {

const int NUM_PINOS = 40;   // o ESP32 tem GPIO 0 a 39

struct Pino {
    int nivel = LOW;
    int modo = -1;          // -1: ninguem chamou pinMode ainda
    int escritas = 0;       // quantas vezes digitalWrite foi chamado neste pino
    int analogico = 0;      // o que analogRead devolve (0 a 4095, como o ADC de 12 bits)
    int atenuacao = -1;     // -1: ninguem chamou analogSetPinAttenuation
    bool pwm = false;       // ledcAttach foi chamado?
    uint32_t pwm_freq = 0;  // Hz
    int pwm_bits = 0;       // resolucao do duty
    uint32_t pwm_duty = 0;  // ultimo valor de ledcWrite
};

inline Pino& pino(int numero) {
    static Pino pinos[NUM_PINOS];
    return pinos[numero % NUM_PINOS];
}

// O relogio guarda MICROSSEGUNDOS; millis() e so ele dividido por 1000.
inline unsigned long long& relogio_us() {
    static unsigned long long agora_us = 0;
    return agora_us;
}

// Para os testes: espiar e mexer no mundo externo.
inline int nivel(int numero) { return pino(numero).nivel; }
inline int modo(int numero) { return pino(numero).modo; }
inline int escritas(int numero) { return pino(numero).escritas; }
inline void poe_nivel(int numero, int nivel) { pino(numero).nivel = nivel; }
inline void poe_analogico(int numero, int bruto) { pino(numero).analogico = bruto; }
inline int atenuacao(int numero) { return pino(numero).atenuacao; }
inline bool pwm(int numero) { return pino(numero).pwm; }
inline uint32_t frequencia(int numero) { return pino(numero).pwm_freq; }
inline int bits_pwm(int numero) { return pino(numero).pwm_bits; }
inline uint32_t duty(int numero) { return pino(numero).pwm_duty; }
inline void avanca_tempo(unsigned long ms) { relogio_us() += 1000ull * ms; }
inline void avanca_micros(unsigned long us) { relogio_us() += us; }

// Volta tudo ao estado de "placa recem-ligada".
inline void reinicia() {
    for (int i = 0; i < NUM_PINOS; i++) {
        pino(i) = Pino{};
    }
    relogio_us() = 0;
}

}  // namespace falso

// ---- As funcoes do Arduino --------------------------------------------------
inline void pinMode(int numero, int modo) {
    falso::pino(numero).modo = modo;
    // Com pull-up interno, um pino de entrada solto le HIGH -- igual ao real.
    if (modo == INPUT_PULLUP) {
        falso::pino(numero).nivel = HIGH;
    }
}

inline void digitalWrite(int numero, int nivel) {
    falso::pino(numero).nivel = nivel;
    falso::pino(numero).escritas++;
}

inline int digitalRead(int numero) {
    return falso::pino(numero).nivel;
}

// Como no ESP32, os dois sao de 32 bits e dao a volta (millis em ~49 dias,
// micros em ~71 minutos).
inline unsigned long millis() {
    return static_cast<uint32_t>(falso::relogio_us() / 1000);
}

inline unsigned long micros() {
    return static_cast<uint32_t>(falso::relogio_us());
}

inline void delay(unsigned long ms) {
    falso::relogio_us() += 1000ull * ms;
}

inline void delayMicroseconds(unsigned int us) {
    falso::relogio_us() += us;
}

// ---- ADC e PWM (LEDC) do ESP32, com as assinaturas do core 3.x -------------
inline void analogSetPinAttenuation(int numero, adc_attenuation_t atenuacao) {
    falso::pino(numero).atenuacao = atenuacao;
}

inline uint16_t analogRead(int numero) {
    return static_cast<uint16_t>(falso::pino(numero).analogico);
}

inline bool ledcAttach(int numero, uint32_t freq, uint8_t bits) {
    falso::Pino& p = falso::pino(numero);
    p.pwm = true;
    p.pwm_freq = freq;
    p.pwm_bits = bits;
    p.pwm_duty = 0;
    return true;
}

inline bool ledcWrite(int numero, uint32_t duty) {
    falso::Pino& p = falso::pino(numero);
    if (!p.pwm) {
        return false;   // como no real: sem ledcAttach, nada acontece
    }
    p.pwm_duty = duty;
    return true;
}

// Como no core 3.x: frequencia 0 so zera o duty; senao o pino passa a 10 bits
// e duty 0x1FF (quase 50%), uma onda quadrada na frequencia pedida.
inline uint32_t ledcWriteTone(int numero, uint32_t freq) {
    falso::Pino& p = falso::pino(numero);
    if (!p.pwm) {
        return 0;
    }
    if (freq == 0) {
        p.pwm_duty = 0;
        return 0;
    }
    p.pwm_freq = freq;
    p.pwm_bits = 10;
    p.pwm_duty = 0x1FF;
    return freq;
}

    ex03¶

    Um stub de sensor que devolve uma sequência de leituras, uma por chamada de le(): é assim que se testa "o nível foi 400, depois 300, depois o fio soltou". Complete o SensorSequencia:

    • adiciona(valor) e adiciona_invalida() põem leituras no fim da sequência (no máximo 8; além disso, ignore);
    • cada le() devolve a próxima leitura; depois da última, repete a última;
    • sem nenhuma leitura programada, le() devolve {0, false};
    • lidas() diz quantas vezes le() foi chamado.

Por isso o índice e o contador estão marcados mutable: é a palavra do C++ para "este campo pode mudar até num método const". Use-a com parcimônia: aqui ela é honesta, porque o índice é detalhe do dublê, não estado do "sensor".

s.adiciona(400);
s.adiciona_invalida();
s.le();    // -> {400, true}
s.le();    // -> {0, false}
s.le();    // -> {0, false}   (repete a ultima)

    EstadoBotao estado_;

        unsigned long agora = millis();

        switch (estado_) {
            case EstadoBotao::Solto:
                if (lido) {
                    estado_ = EstadoBotao::TalvezApertado;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezApertado:
                if (!lido) {
                    estado_ = EstadoBotao::Solto;          // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Apertado;       // firme: e clique

                break;

                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezSolto:

                    estado_ = EstadoBotao::Apertado;       // era quique

                    estado_ = EstadoBotao::Solto;

// desliga_acima (a histerese: no meio, fica como esta). Se a bomba ficar
// ligada por tempo_max_ms sem encher, a caixa nao esta recebendo agua (bomba

// Leitura invalida com a bomba ligada: desliga, por seguranca.
class ControleBomba {
public:
    enum class Estado { Parada, Bombeando, Bloqueada };

    struct Config {
        int32_t liga_abaixo;
        int32_t desliga_acima;
        uint32_t tempo_max_ms;
    };

private:
    const ISensor& nivel_;
    ISaida& bomba_;
    ISaida& alarme_;
    const IRelogio& relogio_;
    const Config config_;
    Estado estado_;
    uint32_t ligou_em_ms_;

public:
    ControleBomba(const ISensor& nivel, ISaida& bomba, ISaida& alarme,
                  const IRelogio& relogio, Config config)
        : nivel_(nivel), bomba_(bomba), alarme_(alarme), relogio_(relogio),
          config_(config), estado_(Estado::Parada), ligou_em_ms_(0) {}

    // Chame a cada volta do loop().
    void atualiza() {
        Leitura nivel = nivel_.le();
        uint32_t agora = relogio_.agora_ms();

        switch (estado_) {
        case Estado::Parada:
            if (nivel.valida && nivel.valor <= config_.liga_abaixo) {
                bomba_.liga();
                ligou_em_ms_ = agora;
                estado_ = Estado::Bombeando;
            }
            break;
        case Estado::Bombeando:
            if (!nivel.valida || nivel.valor >= config_.desliga_acima) {
                bomba_.desliga();
                estado_ = Estado::Parada;
            } else if (agora - ligou_em_ms_ >= config_.tempo_max_ms) {
                bomba_.desliga();
                alarme_.liga();
                estado_ = Estado::Bloqueada;
            }
            break;
        case Estado::Bloqueada:
            break;      // so rearma() tira daqui
        }
    }

    // O operador conferiu a bomba e mandou tentar de novo.
    void rearma() {
        if (estado_ == Estado::Bloqueada) {
            alarme_.desliga();
            estado_ = Estado::Parada;
        }
    }

    Estado estado() const { return estado_; }

    static const int32_t BRUTO_CHEIA = 3690;
    static const int32_t MM_CHEIA = 2000;
    static const int32_t BRUTO_FIO_SOLTO = 200;   // abaixo: invalida

        if (bruto < BRUTO_FIO_SOLTO || bruto > BRUTO_CURTO) {

        int32_t num = (bruto - BRUTO_VAZIA) * MM_CHEIA;
        int32_t den = BRUTO_CHEIA - BRUTO_VAZIA;
        int32_t mm = (num + ((num >= 0) ? den / 2 : -den / 2)) / den;
        if (mm < 0) {
            mm = 0;
        }
        if (mm > MM_CHEIA) {
            mm = MM_CHEIA;
        }
        return {mm, true};
    }

    Um registro sem mexer na lógica. Na bancada, quer-se saber quando a bomba ligou e desligou, para medir quanto tempo a caixa leva para encher. A tentação é pôr Serial.println dentro da ControleBomba. Não faça isso: a lógica passaria a depender do Serial, e os testes do computador quebrariam.

    Em vez disso, complete a SaidaComRegistro (em saida_com_registro.h): uma ISaida que embrulha outra ISaida. O liga() dela chama o liga() da saída real e imprime, por exemplo, bomba liga em 1234 ms (com a hora do IRelogio); o desliga() faz o mesmo; o ligada() pergunta à saída real. Depois, no sketch.ino, crie uma SaidaComRegistro para a bomba e outra para o alarme, e entregue essas à ControleBomba. Uma classe que embrulha outra de mesma interface para acrescentar um comportamento é o padrão Decorator (GoF).

    Critério de pronto:

    1. Ao ligar, o Serial mostra uma linha bomba liga em ... ms, com a hora em milissegundos.
    2. No bloqueio, aparecem bomba desliga em ... ms e alarme liga em ... ms, e as duas horas são a da partida mais 15 000 ms.
    3. O comportamento da bomba (LEDs, -> bombeando, -> BLOQUEADA) é o mesmo do wk01.
    4. controle_bomba.h não foi alterado: a diferença está só no sketch.ino e no arquivo novo.

    Projeto inicial — crie um projeto ESP32 no Wokwi e cole cada aba no arquivo de mesmo nome (veja como montar o projeto).

    // botao_filtrado.h -- debounce como maquina de estados (capitulo 4).
#ifndef BOTAO_FILTRADO_H
#define BOTAO_FILTRADO_H

#include <Arduino.h>

// Quanto tempo a leitura precisa ficar parada para ser levada a serio.
const unsigned long ESTAVEL_MS = 30;

enum class EstadoBotao { Solto, TalvezApertado, Apertado, TalvezSolto };

class BotaoFiltrado {
private:
    const int pino_;
    EstadoBotao estado_;
    unsigned long desde_ms_;   // quando a leitura mudou pela ultima vez
    bool clique_;              // houve clique NESTA volta?

public:
    explicit BotaoFiltrado(int pino)
        : pino_(pino), estado_(EstadoBotao::Solto), desde_ms_(0), clique_(false) {}

    void inicia() {
        pinMode(pino_, INPUT_PULLUP);
    }

    // Chame uma vez por volta do loop(), antes de perguntar qualquer coisa.
    void atualiza() {
        bool lido = (digitalRead(pino_) == LOW);
        unsigned long agora = millis();
        clique_ = false;

        switch (estado_) {
            case EstadoBotao::Solto:
                if (lido) {
                    estado_ = EstadoBotao::TalvezApertado;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezApertado:
                if (!lido) {
                    estado_ = EstadoBotao::Solto;          // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Apertado;       // firme: e clique
                    clique_ = true;
                }
                break;

            case EstadoBotao::Apertado:
                if (!lido) {
                    estado_ = EstadoBotao::TalvezSolto;
                    desde_ms_ = agora;
                }
                break;

            case EstadoBotao::TalvezSolto:
                if (lido) {
                    estado_ = EstadoBotao::Apertado;       // era quique
                } else if (agora - desde_ms_ >= ESTAVEL_MS) {
                    estado_ = EstadoBotao::Solto;
                }
                break;
        }
    }

    bool foi_apertado() const {
        return clique_;
    }

    bool esta_apertado() const {
        return estado_ == EstadoBotao::Apertado || estado_ == EstadoBotao::TalvezSolto;
    }
};

    // controle_bomba.h -- a bomba da caixa d'agua: so conhece interfaces.
#ifndef CONTROLE_BOMBA_H
#define CONTROLE_BOMBA_H

#include <stdint.h>
#include "interfaces.h"

// Liga a bomba quando o nivel cai ate liga_abaixo e desliga quando chega a
// desliga_acima (a histerese: no meio, fica como esta). Se a bomba ficar
// ligada por tempo_max_ms sem encher, a caixa nao esta recebendo agua (bomba
// seca, cano furado): desliga, acende o alarme e so volta depois de rearma().
// Leitura invalida com a bomba ligada: desliga, por seguranca.
class ControleBomba {
public:
    enum class Estado { Parada, Bombeando, Bloqueada };

    struct Config {
        int32_t liga_abaixo;
        int32_t desliga_acima;
        uint32_t tempo_max_ms;
    };

private:
    const ISensor& nivel_;
    ISaida& bomba_;
    ISaida& alarme_;
    const IRelogio& relogio_;
    const Config config_;
    Estado estado_;
    uint32_t ligou_em_ms_;

public:
    ControleBomba(const ISensor& nivel, ISaida& bomba, ISaida& alarme,
                  const IRelogio& relogio, Config config)
        : nivel_(nivel), bomba_(bomba), alarme_(alarme), relogio_(relogio),
          config_(config), estado_(Estado::Parada), ligou_em_ms_(0) {}

    // Chame a cada volta do loop().
    void atualiza() {
        Leitura nivel = nivel_.le();
        uint32_t agora = relogio_.agora_ms();

        switch (estado_) {
        case Estado::Parada:
            if (nivel.valida && nivel.valor <= config_.liga_abaixo) {
                bomba_.liga();
                ligou_em_ms_ = agora;
                estado_ = Estado::Bombeando;
            }
            break;
        case Estado::Bombeando:
            if (!nivel.valida || nivel.valor >= config_.desliga_acima) {
                bomba_.desliga();
                estado_ = Estado::Parada;
            } else if (agora - ligou_em_ms_ >= config_.tempo_max_ms) {
                bomba_.desliga();
                alarme_.liga();
                estado_ = Estado::Bloqueada;
            }
            break;
        case Estado::Bloqueada:
            break;      // so rearma() tira daqui
        }
    }

    // O operador conferiu a bomba e mandou tentar de novo.
    void rearma() {
        if (estado_ == Estado::Bloqueada) {
            alarme_.desliga();
            estado_ = Estado::Parada;
        }
    }

    Estado estado() const { return estado_; }
};

#ifndef SENSOR_NIVEL_H
#define SENSOR_NIVEL_H

#include <Arduino.h>
#include "interfaces.h"

class SensorNivel : public ISensor {
private:
    const int pino_;

    // Calibracao de dois pontos: leitura 410 = caixa vazia, 3690 = 2000 mm.
    static const int32_t BRUTO_VAZIA = 410;
    static const int32_t BRUTO_CHEIA = 3690;
    static const int32_t MM_CHEIA = 2000;
    static const int32_t BRUTO_FIO_SOLTO = 200;   // abaixo: invalida
    static const int32_t BRUTO_CURTO = 3900;      // acima: invalida

public:
    explicit SensorNivel(int pino) : pino_(pino) {}

    void inicia() { analogSetPinAttenuation(pino_, ADC_11db); }

    Leitura le() const override {
        int32_t bruto = analogRead(pino_);
        if (bruto < BRUTO_FIO_SOLTO || bruto > BRUTO_CURTO) {
            return {0, false};
        }
        int32_t num = (bruto - BRUTO_VAZIA) * MM_CHEIA;
        int32_t den = BRUTO_CHEIA - BRUTO_VAZIA;
        int32_t mm = (num + ((num >= 0) ? den / 2 : -den / 2)) / den;
        if (mm < 0) {
            mm = 0;
        }
        if (mm > MM_CHEIA) {
            mm = MM_CHEIA;
        }
        return {mm, true};
    }

    int32_t minimo() const override { return 0; }
    int32_t maximo() const override { return MM_CHEIA; }
    const char* unidade() const override { return "mm"; }
};

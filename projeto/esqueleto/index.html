    Esqueleto do projeto¶

    O projeto final parte deste esqueleto: o da bomba da caixa d'água do capítulo 8. A ideia não é que todo projeto controle uma bomba, e sim que todo projeto tenha esta forma: a lógica só conhece interfaces, os testes rodam no computador com dublês, e o sketch.ino é o único lugar que sabe quais peças estão na bancada.

    Pasta O que tem Depende de
    logica/ as interfaces e a lógica do projeto nada do Arduino
    host/ os dublês, os testes e o compila.sh logica/
    wokwi/ as implementações de verdade, o sketch.ino, o diagram.json e uma cópia de logica/ o core Arduino

    Para começar o seu: copie os arquivos para três pastas com esses nomes, troque a bomba pela lógica do seu tema (mantendo as interfaces que servirem e criando as que faltarem) e escreva os testes antes, como no capítulo. A pasta wokwi/ leva uma cópia dos arquivos de logica/ porque o Wokwi quer tudo junto, em abas: edite em logica/ e copie.

    // interfaces.h -- os contratos entre a logica e o hardware.
// Nada aqui inclui <Arduino.h>: a logica que depende so destes tipos compila
// no computador (com dubles) e no ESP32 (com as classes de verdade).
#ifndef INTERFACES_H
#define INTERFACES_H

#include <stdint.h>
#include "leitura.h"

// Um sensor que entrega a grandeza ja convertida, com a faixa e a validade
// (o sensor inteligente do capitulo 7).
class ISensor {
public:
    virtual ~ISensor() = default;
    virtual Leitura le() const = 0;
    virtual int32_t minimo() const = 0;
    virtual int32_t maximo() const = 0;
    virtual const char* unidade() const = 0;
};

// Uma saida de dois estados: LED, rele, sirene.
class ISaida {
public:
    virtual ~ISaida() = default;
    virtual void liga() = 0;
    virtual void desliga() = 0;
    virtual bool ligada() const = 0;
};

// A hora, em milissegundos. Na placa e o millis(); no teste, um relogio que
// so anda quando o teste manda.
class IRelogio {
public:
    virtual ~IRelogio() = default;
    virtual uint32_t agora_ms() const = 0;
};

#endif

#ifndef LEITURA_H
#define LEITURA_H

#include <stdint.h>

// Um numero na unidade do sensor (permil, decimos de grau...) e se ele presta.
struct Leitura {
    int32_t valor;
    bool valida;    // false: fio solto, curto, ou fora da faixa que o sensor mede

    // controle_bomba.h -- a bomba da caixa d'agua: so conhece interfaces.
#ifndef CONTROLE_BOMBA_H
#define CONTROLE_BOMBA_H

#include <stdint.h>
#include "interfaces.h"

// Liga a bomba quando o nivel cai ate liga_abaixo e desliga quando chega a
// desliga_acima (a histerese: no meio, fica como esta). Se a bomba ficar
// ligada por tempo_max_ms sem encher, a caixa nao esta recebendo agua (bomba
// seca, cano furado): desliga, acende o alarme e so volta depois de rearma().
// Leitura invalida com a bomba ligada: desliga, por seguranca.
class ControleBomba {
public:
    enum class Estado { Parada, Bombeando, Bloqueada };

    struct Config {
        int32_t liga_abaixo;
        int32_t desliga_acima;
        uint32_t tempo_max_ms;
    };

private:
    const ISensor& nivel_;
    ISaida& bomba_;
    ISaida& alarme_;
    const IRelogio& relogio_;
    const Config config_;
    Estado estado_;
    uint32_t ligou_em_ms_;

public:
    ControleBomba(const ISensor& nivel, ISaida& bomba, ISaida& alarme,
                  const IRelogio& relogio, Config config)
        : nivel_(nivel), bomba_(bomba), alarme_(alarme), relogio_(relogio),
          config_(config), estado_(Estado::Parada), ligou_em_ms_(0) {}

    // Chame a cada volta do loop().
    void atualiza() {
        Leitura nivel = nivel_.le();
        uint32_t agora = relogio_.agora_ms();

        switch (estado_) {
        case Estado::Parada:
            if (nivel.valida && nivel.valor <= config_.liga_abaixo) {
                bomba_.liga();
                ligou_em_ms_ = agora;
                estado_ = Estado::Bombeando;
            }
            break;
        case Estado::Bombeando:
            if (!nivel.valida || nivel.valor >= config_.desliga_acima) {
                bomba_.desliga();
                estado_ = Estado::Parada;
            } else if (agora - ligou_em_ms_ >= config_.tempo_max_ms) {
                bomba_.desliga();
                alarme_.liga();
                estado_ = Estado::Bloqueada;
            }
            break;
        case Estado::Bloqueada:
            break;      // so rearma() tira daqui
        }
    }

    // O operador conferiu a bomba e mandou tentar de novo.
    void rearma() {
        if (estado_ == Estado::Bloqueada) {
            alarme_.desliga();
            estado_ = Estado::Parada;
        }
    }

    Estado estado() const { return estado_; }
};

#endif

    Para rodar os testes: ./compila.sh (precisa de g++). A saída esperada é Todos os testes passaram!.

    // Testes da ControleBomba no computador: sensor, saidas e relogio sao dubles.
// Rode com ./compila.sh
#include <cassert>
#include <iostream>
#include "controle_bomba.h"
#include "duplos.h"

const ControleBomba::Config CONFIG = {500, 1800, 10u * 60 * 1000};   // 10 minutos

void enche_e_esvazia() {
    SensorProgramado nivel(0, 2000);
    RelogioFalso relogio;
    SaidaEspia bomba(relogio);
    SaidaEspia alarme(relogio);
    ControleBomba controle(nivel, bomba, alarme, relogio, CONFIG);

    nivel.poe(1200);                 // no meio: nada acontece
    controle.atualiza();
    assert(bomba.quantos() == 0);

    nivel.poe(500);                  // chegou ao minimo: liga
    relogio.avanca(1000);
    controle.atualiza();
    assert(bomba.ligada());
    assert(controle.estado() == ControleBomba::Estado::Bombeando);

    nivel.poe(1200);                 // subindo: continua ligada (histerese)
    relogio.avanca(60000);
    controle.atualiza();
    assert(bomba.ligada());

    nivel.poe(1800);                 // cheia: desliga
    relogio.avanca(60000);
    controle.atualiza();
    assert(!bomba.ligada());

    assert(bomba.quantos() == 2);
    assert(bomba.evento(0).ligou && bomba.evento(0).em_ms == 1000);
    assert(!bomba.evento(1).ligou && bomba.evento(1).em_ms == 121000);
    assert(alarme.quantos() == 0);
}

void bomba_seca() {
    SensorProgramado nivel(0, 2000);
    RelogioFalso relogio;
    SaidaEspia bomba(relogio);
    SaidaEspia alarme(relogio);
    ControleBomba controle(nivel, bomba, alarme, relogio, CONFIG);

    nivel.poe(400);
    controle.atualiza();             // liga em t = 0
    relogio.avanca(10u * 60 * 1000 - 1);
    controle.atualiza();             // 9 min 59,999 s: ainda tentando
    assert(bomba.ligada());
    relogio.avanca(1);
    controle.atualiza();             // 10 min: desiste
    assert(!bomba.ligada());
    assert(alarme.ligada());
    assert(controle.estado() == ControleBomba::Estado::Bloqueada);

    nivel.poe(100);                  // bloqueada: nem com a caixa vazia religa
    controle.atualiza();
    assert(!bomba.ligada());

    controle.rearma();
    assert(!alarme.ligada());
    controle.atualiza();             // rearmada: tenta de novo
    assert(bomba.ligada());
}

void sensor_quebrado() {
    SensorProgramado nivel(0, 2000);
    RelogioFalso relogio;
    SaidaEspia bomba(relogio);
    SaidaEspia alarme(relogio);
    ControleBomba controle(nivel, bomba, alarme, relogio, CONFIG);

    nivel.poe_invalida();            // parada + invalida: nao liga
    controle.atualiza();
    assert(!bomba.ligada());

    nivel.poe(300);
    controle.atualiza();
    assert(bomba.ligada());
    nivel.poe_invalida();            // ligada + invalida: desliga
    controle.atualiza();
    assert(!bomba.ligada());
    assert(!alarme.ligada());
}

void ordem_exata_com_mock() {
    SensorProgramado nivel(0, 2000);
    RelogioFalso relogio;
    SaidaMock bomba;
    SaidaEspia alarme(relogio);
    ControleBomba controle(nivel, bomba, alarme, relogio, CONFIG);

    bomba.espera_liga();
    bomba.espera_desliga();

    nivel.poe(400);
    controle.atualiza();
    controle.atualiza();             // atualizar de novo NAO pode ligar outra vez
    nivel.poe(1900);
    controle.atualiza();
    controle.atualiza();
    assert(bomba.conferiu());
}

int main() {
    enche_e_esvazia();
    bomba_seca();
    sensor_quebrado();
    ordem_exata_com_mock();
    std::cout << "Todos os testes passaram!\n";
    return 0;
}

    // duplos.h -- os dubles de teste do capitulo 8. Nenhum usa heap: listas sao
// arrays fixos, como no firmware.
#ifndef DUPLOS_H
#define DUPLOS_H

#include <stdint.h>
#include "../logica/interfaces.h"

// STUB: devolve o que o teste programou. Nao tem logica nenhuma.
class SensorProgramado : public ISensor {
private:
    Leitura leitura_;
    int32_t minimo_;
    int32_t maximo_;

public:
    SensorProgramado(int32_t minimo, int32_t maximo)
        : leitura_{0, true}, minimo_(minimo), maximo_(maximo) {}

    void poe(int32_t valor) { leitura_ = {valor, true}; }
    void poe_invalida() { leitura_ = {0, false}; }

    Leitura le() const override { return leitura_; }
    int32_t minimo() const override { return minimo_; }
    int32_t maximo() const override { return maximo_; }
    const char* unidade() const override { return "mm"; }
};

// FAKE: um relogio que funciona de verdade, mas so anda quando o teste manda.
class RelogioFalso : public IRelogio {
private:
    uint32_t agora_ms_;

public:
    RelogioFalso() : agora_ms_(0) {}

    void avanca(uint32_t ms) { agora_ms_ += ms; }
    uint32_t agora_ms() const override { return agora_ms_; }
};

// SPY: funciona como uma saida e anota cada mudanca, com a hora.
class SaidaEspia : public ISaida {
public:
    struct Evento {
        bool ligou;
        uint32_t em_ms;
    };

private:
    static const int MAX = 16;
    const IRelogio& relogio_;
    Evento eventos_[MAX];
    int quantos_;
    bool ligada_;

    void anota(bool ligou) {
        if (quantos_ < MAX) {
            eventos_[quantos_] = {ligou, relogio_.agora_ms()};
            quantos_++;
        }
    }

public:
    explicit SaidaEspia(const IRelogio& relogio)
        : relogio_(relogio), eventos_{}, quantos_(0), ligada_(false) {}

    void liga() override { ligada_ = true; anota(true); }
    void desliga() override { ligada_ = false; anota(false); }
    bool ligada() const override { return ligada_; }

    int quantos() const { return quantos_; }
    Evento evento(int i) const { return eventos_[i]; }
};

// MOCK: sabe de antemao o que deveria acontecer e confere a cada chamada.
class SaidaMock : public ISaida {
private:
    static const int MAX = 8;
    bool esperado_[MAX];     // true = liga, false = desliga
    int esperados_;
    int recebidos_;
    int primeiro_erro_;      // -1: nenhum
    bool ligada_;

    void recebe(bool ligou) {
        if (primeiro_erro_ < 0 && (recebidos_ >= esperados_ || esperado_[recebidos_] != ligou)) {
            primeiro_erro_ = recebidos_;
        }
        recebidos_++;
        ligada_ = ligou;
    }

public:
    SaidaMock() : esperado_{}, esperados_(0), recebidos_(0), primeiro_erro_(-1), ligada_(false) {}

    void espera_liga() { if (esperados_ < MAX) { esperado_[esperados_++] = true; } }
    void espera_desliga() { if (esperados_ < MAX) { esperado_[esperados_++] = false; } }

    void liga() override { recebe(true); }
    void desliga() override { recebe(false); }
    bool ligada() const override { return ligada_; }

    // Tudo o que era esperado aconteceu, na ordem, e nada mais?
    bool conferiu() const { return primeiro_erro_ < 0 && recebidos_ == esperados_; }
    int primeiro_erro() const { return primeiro_erro_; }
    int recebidos() const { return recebidos_; }
};

#endif

    // Capitulo 8 -- a bomba da caixa d'agua. A ControleBomba so conhece interfaces;
// este arquivo e o unico lugar que sabe quais pecas estao na bancada.
// (Para a demonstracao, a bomba desiste depois de 15 s; numa caixa real, minutos.)
#include "controle_bomba.h"
#include "sensor_nivel.h"
#include "saida_digital.h"
#include "relogio_arduino.h"
#include "botao_filtrado.h"
#include "temporizador.h"

SensorNivel nivel(34);          // potenciometro no papel de sensor de nivel
SaidaDigital bomba(4);          // modulo rele, ativo em nivel alto
SaidaDigital alarme(26);        // LED vermelho
RelogioArduino relogio;
BotaoFiltrado rearme(32);       // botao A
ControleBomba controle(nivel, bomba, alarme, relogio, {500, 1800, 15000});
Temporizador um_segundo(1000);

ControleBomba::Estado anterior = ControleBomba::Estado::Parada;

const char* nome(ControleBomba::Estado e) {
    switch (e) {
    case ControleBomba::Estado::Parada:    return "parada";
    case ControleBomba::Estado::Bombeando: return "bombeando";
    case ControleBomba::Estado::Bloqueada: return "BLOQUEADA";
    }
    return "?";
}

void setup() {
    Serial.begin(115200);
    nivel.inicia();
    bomba.inicia();
    alarme.inicia();
    rearme.inicia();
    um_segundo.reinicia();
    Serial.println("bomba da caixa d'agua");
}

void loop() {
    rearme.atualiza();
    if (rearme.foi_apertado()) {
        controle.rearma();
    }
    controle.atualiza();

    if (controle.estado() != anterior) {
        Serial.print("-> ");
        Serial.println(nome(controle.estado()));
        anterior = controle.estado();
    }
    if (um_segundo.expirou()) {
        Leitura l = nivel.le();
        if (l.valida) {
            Serial.print(l.valor);
            Serial.println(" mm");
        } else {
            Serial.println("sensor invalido");
        }
    }
}

    // sensor_nivel.h -- o sensor de nivel do capitulo 7, agora implementando ISensor.
// Na bancada do Wokwi, o "sensor" e o potenciometro do pino 34.
#ifndef SENSOR_NIVEL_H
#define SENSOR_NIVEL_H

#include <Arduino.h>
#include "interfaces.h"

class SensorNivel : public ISensor {
private:
    const int pino_;

    // Calibracao de dois pontos: leitura 410 = caixa vazia, 3690 = 2000 mm.
    static const int32_t BRUTO_VAZIA = 410;
    static const int32_t BRUTO_CHEIA = 3690;
    static const int32_t MM_CHEIA = 2000;
    static const int32_t BRUTO_FIO_SOLTO = 200;   // abaixo: invalida
    static const int32_t BRUTO_CURTO = 3900;      // acima: invalida

public:
    explicit SensorNivel(int pino) : pino_(pino) {}

    void inicia() { analogSetPinAttenuation(pino_, ADC_11db); }

    Leitura le() const override {
        int32_t bruto = analogRead(pino_);
        if (bruto < BRUTO_FIO_SOLTO || bruto > BRUTO_CURTO) {
            return {0, false};
        }
        int32_t num = (bruto - BRUTO_VAZIA) * MM_CHEIA;
        int32_t den = BRUTO_CHEIA - BRUTO_VAZIA;
        int32_t mm = (num + ((num >= 0) ? den / 2 : -den / 2)) / den;
        if (mm < 0) {
            mm = 0;
        }
        if (mm > MM_CHEIA) {
            mm = MM_CHEIA;
        }
        return {mm, true};
    }

    int32_t minimo() const override { return 0; }
    int32_t maximo() const override { return MM_CHEIA; }
    const char* unidade() const override { return "mm"; }
};

#endif

    // saida_digital.h -- uma ISaida num pino digital: LED, modulo rele, sirene.
#ifndef SAIDA_DIGITAL_H
#define SAIDA_DIGITAL_H

#include <Arduino.h>
#include "interfaces.h"

class SaidaDigital : public ISaida {
private:
    const int pino_;
    const bool ativo_alto_;     // false: liga com LOW (muitos modulos rele sao assim)
    bool ligada_;

public:
    explicit SaidaDigital(int pino, bool ativo_alto = true)
        : pino_(pino), ativo_alto_(ativo_alto), ligada_(false) {}

    void inicia() {
        pinMode(pino_, OUTPUT);
        desliga();
    }

    void liga() override {
        digitalWrite(pino_, ativo_alto_ ? HIGH : LOW);
        ligada_ = true;
    }

    void desliga() override {
        digitalWrite(pino_, ativo_alto_ ? LOW : HIGH);
        ligada_ = false;
    }

    bool ligada() const override { return ligada_; }
};

#endif
